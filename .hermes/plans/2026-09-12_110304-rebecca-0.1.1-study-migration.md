```kotlin
package top.guangyiliushan.rebecca.core.logic

import kotlin.random.Random
import top.guangyiliushan.rebecca.core.model.QuizMode
import top.guangyiliushan.rebecca.core.model.Sense

/** 一题：题干来自 target（definition），四选项为 lemma 形态（UI 层把 lemmaId 换 form）。 */
data class QuizQuestion(
    val target: Sense,
    val optionLemmaIds: List<String>,
    val correctIndex: Int,
)

/**
 * 构建一题：target + (optionCount-1) 个干扰项。
 * 干扰项排除与 target 同 lemmaId 的 Sense（同词形多义项会造成重复选项）；
 * pool 中可用干扰不足时返回 null（调用方结束会话）。
 */
fun buildQuestion(target: Sense, pool: List<Sense>, optionCount: Int = 4, rng: Random = Random.Default): QuizQuestion? {
    val distractors = pool.filter { it.lemmaId != target.lemmaId }.toMutableList()
    if (distractors.size < optionCount - 1) return null
    distractors.shuffle(rng)
    val chosen = distractors.take(optionCount - 1)
    val optionIds = (listOf(target) + chosen).map { it.lemmaId }.shuffled(rng)
    return QuizQuestion(target, optionIds, optionIds.indexOf(target.lemmaId))
}

fun isCorrect(question: QuizQuestion, chosenIndex: Int): Boolean = chosenIndex == question.correctIndex

/** 三模式候选筛选（Q6 裁定语义）。MIXED = 新词:到期 1:1 交错，rng 只决定组内顺序。 */
fun selectCandidates(mode: QuizMode, new: List<Sense>, due: List<Sense>, limit: Int, rng: Random): List<Sense> = when (mode) {
    QuizMode.LEARN -> new.shuffled(rng).take(limit)
    QuizMode.REVIEW -> due.shuffled(rng).take(limit)
    QuizMode.MIXED -> {
        val n = new.shuffled(rng)
        val d = due.shuffled(rng)
        val per = minOf(n.size, d.size, (limit + 1) / 2)
        buildList {
            repeat(per) { i -> add(n[i]); add(d[i]) }
        }.take(limit)
    }
}
```

```kotlin
package top.guangyiliushan.rebecca.core.logic

import kotlin.time.Instant
import top.guangyiliushan.rebecca.core.model.MasteryEvent

/**
 * 连续学习天数（Q8 裁定 a：中枢核心数字）。
 * v0 以 UTC 日切分（epoch 86400s）；时区敏感的正确日历日归 0.2.3 统计版（事件流聚合）。
 * 语义：今天有事件从今天数起，否则从昨天数起（今天没学不算断）；遇断即停。
 */
fun currentStreak(events: List<MasteryEvent>, now: Instant): Int {
    fun dayOf(t: Instant): Long = t.epochSeconds / 86_400
    val days = events.map { dayOf(it.occurredAt) }.toSet()
    var d = dayOf(now)
    if (d !in days) d -= 1
    var streak = 0
    while (d in days) { streak++; d-- }
    return streak
}
```

```kotlin
data class QuizUiState(
    val mode: QuizMode,
    val question: QuizQuestion? = null,       // null = 无题可练（空态）
    val index: Int = 0, val total: Int = 0,
    val selectedIndex: Int? = null,
    val wrongIndices: Set<Int> = emptySet(),  // 本题已错选项（标红 + 可继续选，Q7：仅内存）
    val answeredCorrect: Boolean = false,     // 选对 → 详解卡态
    val correctCount: Int = 0,
    val finished: Boolean = false,
    val elapsedSeconds: Long = 0,
    val optionForms: List<String> = emptyList(),  // lemmaId → form 的 UI 侧解析结果
)

sealed interface QuizUiEvent {
    data class OptionChosen(val index: Int) : QuizUiEvent
    data object Next : QuizUiEvent
    data object Retry : QuizUiEvent
    data object BackToHub : QuizUiEvent   // 由 App.kt 导航层处理
}

class QuizViewModel(
    private val mode: QuizMode,
    private val nowSeconds: () -> Long = { System.currentTimeMillis() / 1000 },
) : ViewModel() {
    // 会话组装（R1：只编排，判分/选词在 core）：
    // init：new = allSenses - mastered；due = dueSenses()；candidates = selectCandidates(mode, new, due, 20, Random)
    //       队列 = candidates 逐个 buildQuestion(target, allSenses)；buildQuestion 返回 null 即停
    // OptionChosen：isCorrect → answeredCorrect=true, correctCount++；错 → wrongIndices += index（不改正确计数，可继续选）
    // Next：下一题（buildQuestion 队列取完或 null → finished=true）
    // Retry：重跑 init 装配
    // elapsed：会话开始时记录 start，finished 时结算
}
```

```kotlin
    private fun vm(mode: QuizMode) = QuizViewModel(mode)

    fun wrongChoice_allowsRetryAndMarksWrong() {

        assertFalse(vm.uiState.value.answeredCorrect)
        vm.onEvent(QuizUiEvent.OptionChosen(correct))
```

# 风险与权衡

1. **Route sealed 穷尽 when**：加 `Route.Quiz` 成员会同时编译失败 App.kt/fragmentName/routeFromFragmentName 三处——A5 已列全部改动点与临时占位分支（Phase C 替换）。
2. **demo 种子改动连锁**：mastery 覆盖从 28→20、dueSenses 加截止——中枢数字、契约测试、QuizViewModel 初始态全部与 A4 种子绑定；任何一处改种子参数须同步 StudyHubViewModelTest/QuizViewModelTest 预期值。
3. **QuizMode 序列化**：需 core 加 kotlinx-serialization plugin（若缺）。Web URL 恢复经 `routeFromFragmentName` 的 fallback 到 LEARN——畸形 fragment 不崩（F-L5 精神）。
4. **streak 的 UTC 日切分**：epoch 86400s 切日在非 UTC 时区边界会错一天；demo 无时区概念，0.2.3 事件流聚合版接 kotlinx-datetime 时正式处理（计划内已注）。
5. **选项 form 解析在 UI 层**：`optionLemmaIds` → form 经 `ContentDictionaryRepository.lemma()`，core 判分只对 index——UI 解析失败（lemma 缺失）时选项显示 lemmaId 原值兜底，不崩。
6. **VM 双实现迁移面（Q5 已裁定接受）**：0.2.0 下沉时 QuizViewModel 重写为状态机快照渲染，但 `buildQuestion`/`isCorrect`/`selectCandidates` 纯函数零重写。
7. **新增 pattern 必须走 showcase 门禁**：StatCard/EmptyState 若漏注册或 preview 数 <3，ShowcaseRegistryTest 直接失败——B2 已含注册与 COMPONENTS.md 重生成。
8. **TimeSource 实验 API**：elapsed 用注入 `nowSeconds: () -> Long` 而非 kotlin.time.TimeSource，避免 @ExperimentalTime 污染 public API。
9. **同词形多义项**：题干=definition、选项=form 的 v0 形状下，distractor 排除同 lemma 已保证选项唯一（buildQuestion 语义）；题干方向（form→definition）留 0.2.0 出题引擎扩题型。
